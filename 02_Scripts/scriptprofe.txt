	FUNCTION STOCK_DISPONIBLE(
		p_producto_id in number
		)return number
		is
		v_stock PRODUCTO.stock%TYPE;
BEGIN 
	SELECT stock 
	INTO v_stock
	FROM PRODUCTO
	WHERE producto_id = p_producto_id;
	
	return v_stock;
	
	exception 
	when no_data_found THEN 
	RAISE_APPLICATION_ERROR(-20001, 'PRODUCTO INEXISTE2
	END STOCK_DISPONIBLE;
